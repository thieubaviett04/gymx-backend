# KẾ HOẠCH TRIỂN KHAI VÀ CHI TIẾT MÃ NGUỒN BACKEND (PHIÊN BẢN ENTERPRISE)
## MODULE 4.23: QUẢN LÝ GÓI TẬP (MEMBERSHIP PACKAGES MANAGEMENT - P6.2)

> **Cập nhật kiến trúc chuẩn Enterprise:**
> 1. **Chống Race Condition sinh mã (`PackageCode`):** Cơ chế Retry loop kết hợp bắt `DbUpdateException` (Postgres Unique Violation code `23505`) tự động tái sinh mã mới an toàn trong môi trường đa luồng.
> 2. **Tối ưu Index truy vấn (Tầng Infrastructure):** Chuyển từ `.ToLower().Contains()` sang `EF.Functions.ILike(...)` của PostgreSQL để bảo toàn 100% Database B-Tree / GIN Index, triệt tiêu Full Table Scan.
> 3. **Truy vết phả hệ gói tập (Luồng A-3):** Bổ sung `PreviousPackageId` (Self-referencing Foreign Key) để giữ dấu vết lịch sử thay đổi giá / thời hạn của gói tập theo thời gian.
> 4. **Filtered Unique Index cho Xóa mềm (Soft Delete):** Cấu hình `.HasFilter("\"IsDeleted\" = false")` để PostgreSQL chỉ kiểm tra Unique trên các bản ghi chưa bị xóa mềm.

## I. TỔNG QUAN KIẾN TRÚC & MA TRẬN NGHIỆP VỤ (BUSINESS MATRIX)

| Nghiệp vụ | Quy định & Xử lý logic | Xử lý kiến trúc Enterprise |
| :--- | :--- | :--- |
| **Mã gói tập (`PackageCode`)** | Tự sinh **6 ký tự ngẫu nhiên chữ in hoa và số** (A-Z, 0-9). Không cho phép sửa. | Filtered Unique Index + Retry Loop khi gặp Concurrency Collision. |
| **Ngày bắt đầu (`EffectiveStartDate`)** | Mặc định hôm nay. **Không cho phép chọn ngày quá khứ** (`>= Today`). | FluentValidation check `EffectiveStartDate >= Today`. |
| **Kích hoạt tự động** | Nếu chọn ngày tương lai (`> Today`) $\rightarrow$ Trạng thái tự động là **`Chưa hoạt động` (`INACTIVE`)**. | Xử lý tự động trong Domain Entity `MembershipPackage.Create`. |
| **Trạng thái gói (`Status`)** | `ACTIVE` (Hoạt động), `INACTIVE` (Chưa hoạt động), `DISCONTINUED` (Ngừng kinh doanh). | Hằng số `PackageStatus` Constants. |
| **SỬA gói đang có người dùng (A-3 & A-4)** | Nếu có hội viên đang sử dụng (`member_packages` có `Status == ACTIVE`):<br>• **A-3 (Lưu mới):** Đóng gói cũ sang `DISCONTINUED` + set `EffectiveEndDate = Today`; Tạo gói mới lưu `PreviousPackageId` trỏ về gói cũ.<br>• **A-4 (Chỉ ngừng kinh doanh):** Chỉ cập nhật gói hiện tại sang `DISCONTINUED`. | Trả về `409 Conflict` (`Package.InUse`) nếu gửi sửa thông thường; hỗ trợ tham số `Mode`. |
| **XÓA gói đang có người dùng (A-2)** | **Chặn xóa mềm** nếu đang có người dùng. Đề xuất chuyển trạng thái `Ngừng kinh doanh`. Nếu không có ai dùng $\rightarrow$ Soft Delete. | `409 Conflict` (`Package.CannotDeleteInUse`). |
| **Tìm kiếm & Lọc (4.23.4)** | • Tìm kiếm theo `PackageCode` hoặc `Name`<br>• Lọc thời hạn: `<= 1 tháng`, `<= 6 tháng`, `<= 12 tháng`<br>• Sắp xếp giá: `price_asc`, `price_desc`<br>• Lọc trạng thái: `ACTIVE`, `INACTIVE`, `DISCONTINUED` | Sử dụng `EF.Functions.ILike` trên PostgreSQL để bảo toàn Index. |

```csharp
using System.Security.Cryptography;
using GymX.Domain.Common.Models;
using GymX.Domain.Constants;
using GymX.Domain.Contracts;

namespace GymX.Domain.Entities.Packages;

public class MembershipPackage : EntityAuditBase, ISoftDelete
{
    public string PackageCode { get; private set; } = string.Empty;
    public string Name { get; private set; } = string.Empty;
    public decimal Price { get; private set; }
    public int DurationMonths { get; private set; }
    public int DurationDays { get; private set; }
    public int? MaxCheckins { get; private set; }
    public string? Description { get; private set; }
    public string Status { get; private set; } = PackageStatus.Active;
    public DateOnly EffectiveStartDate { get; private set; }
    public DateOnly? EffectiveEndDate { get; private set; }

    /// <summary>
    /// ID của gói tập tiền nhiệm (nếu gói này được sinh ra từ luồng A-3 sửa giá/thời hạn khi gói cũ đang có người dùng)
    /// </summary>
    public Guid? PreviousPackageId { get; private set; }
    public virtual MembershipPackage? PreviousPackage { get; private set; }

    public bool IsDeleted { get; set; }
    public DateTime? DeletedDate { get; set; }

    protected MembershipPackage() { }

    /// <summary>
    /// Sinh ngẫu nhiên 6 ký tự gồm chữ in hoa và chữ số (A-Z, 0-9)
    /// </summary>
    public static string GenerateRandomCode()
    {
        const string chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
        var buffer = new char[6];
        using var rng = RandomNumberGenerator.Create();
        var bytes = new byte[6];
        rng.GetBytes(bytes);

        for (int i = 0; i < 6; i++)
        {
            buffer[i] = chars[bytes[i] % chars.Length];
        }

        return new string(buffer);
    }

    /// <summary>
    /// Khởi tạo gói tập mới: tự sinh mã 6 ký tự, nếu ngày bắt đầu ở tương lai thì tự chuyển thành INACTIVE
    /// </summary>
    public static MembershipPackage Create(
        string name,
        decimal price,
        int durationMonths,
        DateOnly effectiveStartDate,
        DateOnly? effectiveEndDate = null,
        int? maxCheckins = null,
        string? description = null,
        string? customCode = null,
        Guid? previousPackageId = null)
    {
        var today = DateOnly.FromDateTime(DateTime.UtcNow);
        var initialStatus = effectiveStartDate > today ? PackageStatus.Inactive : PackageStatus.Active;

        return new MembershipPackage
        {
            PackageCode = string.IsNullOrWhiteSpace(customCode) ? GenerateRandomCode() : customCode.Trim().ToUpperInvariant(),
            Name = name.Trim(),
            Price = price,
            DurationMonths = durationMonths,
            DurationDays = durationMonths * 30,
            MaxCheckins = maxCheckins,
            Description = string.IsNullOrWhiteSpace(description) ? null : description.Trim(),
            EffectiveStartDate = effectiveStartDate,
            EffectiveEndDate = effectiveEndDate,
            Status = initialStatus,
            PreviousPackageId = previousPackageId
        };
    }

    /// <summary>
    /// Tái sinh mã gói tập mới (dùng trong retry loop khi gặp race condition tại database)
    /// </summary>
    public void RegenerateCode(string newCode)
    {
        PackageCode = newCode.Trim().ToUpperInvariant();
    }

    /// <summary>
    /// Cập nhật thông tin gói tập khi không có ai đang sử dụng
    /// </summary>
    public void UpdateInfo(
        string name,
        decimal price,
        int durationMonths,
        DateOnly effectiveStartDate,
        DateOnly? effectiveEndDate,
        string? status,
        int? maxCheckins = null,
        string? description = null)
    {
        var today = DateOnly.FromDateTime(DateTime.UtcNow);

        Name = name.Trim();
        Price = price;
        DurationMonths = durationMonths;
        DurationDays = durationMonths * 30;
        MaxCheckins = maxCheckins;
        Description = string.IsNullOrWhiteSpace(description) ? null : description.Trim();
        EffectiveStartDate = effectiveStartDate;
        EffectiveEndDate = effectiveEndDate;

        if (effectiveStartDate > today)
        {
            Status = PackageStatus.Inactive;
        }
        else if (!string.IsNullOrWhiteSpace(status) && PackageStatus.IsValid(status))
        {
            Status = status.Trim().ToUpperInvariant();
        }
    }

    /// <summary>
    /// Chuyển trạng thái sang Ngừng kinh doanh và chốt ngày kết thúc
    /// </summary>
    public void Discontinue(DateOnly? endDate = null)
    {
        Status = PackageStatus.Discontinued;
        EffectiveEndDate = endDate ?? DateOnly.FromDateTime(DateTime.UtcNow);
    }

    /// <summary>
    /// Xóa mềm gói tập
    /// </summary>
    public void SoftDelete()
    {
        IsDeleted = true;
        DeletedDate = DateTime.UtcNow;
    }
}
```

```csharp
using GymX.Domain.Entities.Packages;
using Microsoft.EntityFrameworkCore;
using Microsoft.EntityFrameworkCore.Metadata.Builders;

namespace GymX.Infrastructure.Persistence.Configurations;

public class MembershipPackageConfiguration : IEntityTypeConfiguration<MembershipPackage>
{
    public void Configure(EntityTypeBuilder<MembershipPackage> builder)
    {
        builder.ToTable("membership_packages");

        // Filtered Unique Index cho PostgreSQL: Chỉ kiểm tra tính duy nhất trên các bản ghi chưa bị xóa mềm
        builder.HasIndex(p => p.PackageCode)
            .IsUnique()
            .HasFilter("\"IsDeleted\" = false");

        builder.HasIndex(p => p.Status);

        builder.Property(p => p.PackageCode).IsRequired().HasMaxLength(20);
        builder.Property(p => p.Name).IsRequired().HasMaxLength(100);
        builder.Property(p => p.Price).HasColumnType("numeric(12,2)");
        builder.Property(p => p.Status).IsRequired().HasMaxLength(20);

        // Quan hệ phả hệ tự tham chiếu (Self-referencing Relationship)
        builder.HasOne(p => p.PreviousPackage)
            .WithMany()
            .HasForeignKey(p => p.PreviousPackageId)
            .OnDelete(DeleteBehavior.SetNull);
    }
}
```

#### 2.3. `src/GymX.Infrastructure/Repositories/PackageRepository.cs`
> **Điểm nâng cấp:**
> * Thay thế toàn bộ `.ToLower().Contains()` bằng `EF.Functions.ILike(p.PackageCode, $"%{keyword}%")` và `EF.Functions.ILike(p.Name, $"%{keyword}%")` giúp PostgreSQL tận dụng chỉ mục B-tree / GIN, triệt tiêu Full Table Scan.
> * Map thêm thông tin `PreviousPackageCode` vào DTOs.

```csharp
using GymX.Application.Common.Interfaces.Repositories;
using GymX.Application.Modules.Packages.DTOs.Response;
using GymX.Domain.Common.Models;
using GymX.Domain.Constants;
using GymX.Domain.Entities.Packages;
using GymX.Infrastructure.Persistence;
using Microsoft.EntityFrameworkCore;

namespace GymX.Infrastructure.Repositories;

public class PackageRepository(ApplicationDbContext context) 
    : GenericRepository<MembershipPackage>(context), IPackageRepository
{
    public Task<MembershipPackage?> GetByCodeAsync(string packageCode, CancellationToken cancellationToken = default)
        => DbSet.FirstOrDefaultAsync(p => p.PackageCode == packageCode && !p.IsDeleted, cancellationToken);

    public Task<bool> ExistsCodeAsync(string packageCode, CancellationToken cancellationToken = default)
        => DbSet.AnyAsync(p => p.PackageCode == packageCode && !p.IsDeleted, cancellationToken);

    public async Task<bool> IsPackageInUseAsync(Guid packageId, CancellationToken cancellationToken = default)
    {
        var today = DateOnly.FromDateTime(DateTime.UtcNow);
        return await Context.MemberPackages.AnyAsync(mp =>
            mp.PackageId == packageId &&
            mp.Status == MemberPackageStatus.Active &&
            mp.EndDate >= today,
            cancellationToken);
    }

    public async Task<int> CountActiveMembersAsync(Guid packageId, CancellationToken cancellationToken = default)
    {
        var today = DateOnly.FromDateTime(DateTime.UtcNow);
        return await Context.MemberPackages.CountAsync(mp =>
            mp.PackageId == packageId &&
            mp.Status == MemberPackageStatus.Active &&
            mp.EndDate >= today,
            cancellationToken);
    }

    public async Task<PagedResult<PackageSummaryDto>> GetPackagesPagedAsync(
        string? search,
        string? durationFilter,
        string? status,
        string? sortBy,
        int pageIndex,
        int pageSize,
        CancellationToken cancellationToken = default)
    {
        var query = DbSet.AsNoTracking().Where(p => !p.IsDeleted);

        // Lọc từ khóa: Dùng EF.Functions.ILike để PostgreSQL tận dụng Index tối ưu
        if (!string.IsNullOrWhiteSpace(search))
        {
            var keyword = search.Trim();
            query = query.Where(p =>
                EF.Functions.ILike(p.PackageCode, $"%{keyword}%") ||
                EF.Functions.ILike(p.Name, $"%{keyword}%"));
        }

        // Lọc theo thời hạn (UC 4.23.4 A-2)
        if (!string.IsNullOrWhiteSpace(durationFilter))
        {
            var filter = durationFilter.Trim().ToLower();
            query = filter switch
            {
                "<=1" or "1" => query.Where(p => p.DurationMonths <= 1 || p.DurationDays <= 31),
                "<=6" or "6" => query.Where(p => p.DurationMonths <= 6 || p.DurationDays <= 186),
                "<=12" or "12" => query.Where(p => p.DurationMonths <= 12 || p.DurationDays <= 366),
                _ => query
            };
        }

        // Lọc theo trạng thái
        if (!string.IsNullOrWhiteSpace(status))
        {
            var stat = status.Trim().ToUpperInvariant();
            query = query.Where(p => p.Status == stat);
        }

        // Sắp xếp theo giá (UC 4.23.4 A-2)
        query = sortBy?.Trim().ToLower() switch
        {
            "price_asc" or "price-asc" => query.OrderBy(p => p.Price),
            "price_desc" or "price-desc" => query.OrderByDescending(p => p.Price),
            _ => query.OrderByDescending(p => p.CreatedDate)
        };

        var totalCount = await query.LongCountAsync(cancellationToken);
        var today = DateOnly.FromDateTime(DateTime.UtcNow);

        var items = await query
            .Skip((pageIndex - 1) * pageSize)
            .Take(pageSize)
            .Select(p => new PackageSummaryDto
            {
                Id = p.Id,
                PackageCode = p.PackageCode,
                Name = p.Name,
                Price = p.Price,
                DurationMonths = p.DurationMonths,
                DurationDays = p.DurationDays,
                MaxCheckins = p.MaxCheckins,
                Description = p.Description,
                Status = p.Status,
                EffectiveStartDate = p.EffectiveStartDate,
                EffectiveEndDate = p.EffectiveEndDate,
                PreviousPackageId = p.PreviousPackageId,
                PreviousPackageCode = p.PreviousPackage != null ? p.PreviousPackage.PackageCode : null,
                ActiveMembersCount = Context.MemberPackages.Count(mp =>
                    mp.PackageId == p.Id &&
                    mp.Status == MemberPackageStatus.Active &&
                    mp.EndDate >= today),
                CreatedDate = p.CreatedDate
            })
            .ToListAsync(cancellationToken);

        return new PagedResult<PackageSummaryDto>(items, totalCount, pageIndex, pageSize);
    }

    public async Task<PackageDetailDto?> GetPackageDetailByIdAsync(Guid packageId, CancellationToken cancellationToken = default)
    {
        var package = await DbSet.AsNoTracking()
            .Include(p => p.PreviousPackage)
            .FirstOrDefaultAsync(p => p.Id == packageId && !p.IsDeleted, cancellationToken);

        if (package is null) return null;

        var today = DateOnly.FromDateTime(DateTime.UtcNow);
        var activeCount = await Context.MemberPackages.CountAsync(mp =>
            mp.PackageId == packageId &&
            mp.Status == MemberPackageStatus.Active &&
            mp.EndDate >= today,
            cancellationToken);

        return new PackageDetailDto
        {
            Id = package.Id,
            PackageCode = package.PackageCode,
            Name = package.Name,
            Price = package.Price,
            DurationMonths = package.DurationMonths,
            DurationDays = package.DurationDays,
            MaxCheckins = package.MaxCheckins,
            Description = package.Description,
            Status = package.Status,
            EffectiveStartDate = package.EffectiveStartDate,
            EffectiveEndDate = package.EffectiveEndDate,
            PreviousPackageId = package.PreviousPackageId,
            PreviousPackageCode = package.PreviousPackage?.PackageCode,
            ActiveMembersCount = activeCount,
            CreatedDate = package.CreatedDate,
            LastModifiedDate = package.LastModifiedDate
        };
    }
}
```

#### 3.2. Commands với Cơ chế Retry Loop chống Race Condition

* **`CreatePackageCommandHandler.cs`**
> **Điểm nâng cấp:** Bọc lệnh lưu trong vòng lặp thử lại tối đa 3 lần. Nếu bắt được `DbUpdateException` do trùng mã tại Database, handler tự sinh mã mới và lưu lại an toàn.

```csharp
using GymX.Application.Common.Interfaces;
using GymX.Application.Common.Interfaces.Repositories;
using GymX.Application.Modules.Packages.DTOs.Request;
using GymX.Application.Modules.Packages.DTOs.Response;
using GymX.Domain.Common.Models;
using GymX.Domain.Entities.Packages;
using MediatR;
using Microsoft.EntityFrameworkCore;
using Npgsql;

namespace GymX.Application.Modules.Packages.Commands.CreatePackage;

public record CreatePackageCommand(
    string Name,
    decimal Price,
    int DurationMonths,
    DateOnly EffectiveStartDate,
    DateOnly? EffectiveEndDate,
    int? MaxCheckins,
    string? Description) : IRequest<Result<PackageDetailDto>>
{
    public static CreatePackageCommand FromRequest(CreatePackageRequest req) => new(
        req.Name.Trim(),
        req.Price,
        req.DurationMonths,
        req.EffectiveStartDate,
        req.EffectiveEndDate,
        req.MaxCheckins,
        string.IsNullOrWhiteSpace(req.Description) ? null : req.Description.Trim());
}

public class CreatePackageCommandHandler(
    IPackageRepository packageRepository,
    IUnitOfWork unitOfWork)
    : IRequestHandler<CreatePackageCommand, Result<PackageDetailDto>>
{
    private const int MaxSaveRetries = 3;

    public async Task<Result<PackageDetailDto>> Handle(CreatePackageCommand command, CancellationToken cancellationToken)
    {
        var package = MembershipPackage.Create(
            command.Name,
            command.Price,
            command.DurationMonths,
            command.EffectiveStartDate,
            command.EffectiveEndDate,
            command.MaxCheckins,
            command.Description);

        await packageRepository.AddAsync(package, cancellationToken);

        // Retry loop chống Race Condition khi có 2 request sinh trùng mã cùng mili-giây
        var saveSucceeded = false;
        for (var attempt = 1; attempt <= MaxSaveRetries; attempt++)
        {
            try
            {
                await unitOfWork.SaveChangesAsync(cancellationToken);
                saveSucceeded = true;
                break;
            }
            catch (DbUpdateException ex) when (IsUniqueConstraintViolation(ex) && attempt < MaxSaveRetries)
            {
                // Tự động sinh mã 6 ký tự mới và thử lưu lại
                package.RegenerateCode(MembershipPackage.GenerateRandomCode());
            }
        }

        if (!saveSucceeded)
        {
            return Result.Failure<PackageDetailDto>(
                Error.Conflict("Package.CodeCollision", "Không thể sinh mã gói tập duy nhất. Vui lòng thử lại."));
        }

        var resultDto = new PackageDetailDto
        {
            Id = package.Id,
            PackageCode = package.PackageCode,
            Name = package.Name,
            Price = package.Price,
            DurationMonths = package.DurationMonths,
            DurationDays = package.DurationDays,
            MaxCheckins = package.MaxCheckins,
            Description = package.Description,
            Status = package.Status,
            EffectiveStartDate = package.EffectiveStartDate,
            EffectiveEndDate = package.EffectiveEndDate,
            PreviousPackageId = null,
            PreviousPackageCode = null,
            ActiveMembersCount = 0,
            CreatedDate = package.CreatedDate
        };

        return Result.Success(resultDto);
    }

    private static bool IsUniqueConstraintViolation(DbUpdateException ex)
    {
        return ex.InnerException is PostgresException pgEx && pgEx.SqlState == PostgresErrorCodes.UniqueViolation;
    }
}
```

```csharp
public record UpdatePackageCommand(
    Guid PackageId,
    string Name,
    decimal Price,
    int DurationMonths,
    DateOnly EffectiveStartDate,
    DateOnly? EffectiveEndDate,
    string? Status,
    int? MaxCheckins,
    string? Description,
    string? Mode = null) : IRequest<Result<PackageDetailDto>>
{
    public static UpdatePackageCommand FromRequest(Guid packageId, UpdatePackageRequest req) => new(
        packageId,
        req.Name.Trim(),
        req.Price,
        req.DurationMonths,
        req.EffectiveStartDate,
        req.EffectiveEndDate,
        req.Status?.Trim().ToUpperInvariant(),
        req.MaxCheckins,
        string.IsNullOrWhiteSpace(req.Description) ? null : req.Description.Trim(),
        req.Mode?.Trim().ToUpperInvariant());
}

public class UpdatePackageCommandHandler(
    IPackageRepository packageRepository,
    IUnitOfWork unitOfWork)
    : IRequestHandler<UpdatePackageCommand, Result<PackageDetailDto>>
{
    private const int MaxSaveRetries = 3;

    public async Task<Result<PackageDetailDto>> Handle(UpdatePackageCommand command, CancellationToken cancellationToken)
    {
        var package = await packageRepository.GetByIdAsync(command.PackageId, cancellationToken);
        if (package is null || package.IsDeleted)
        {
            return Result.Failure<PackageDetailDto>(
                Error.NotFound("Package.NotFound", "Không tìm thấy thông tin gói tập."));
        }

        var isBeingUsed = await packageRepository.IsPackageInUseAsync(package.Id, cancellationToken);

        // Trường hợp đang có người dùng (Luồng A-3 & A-4)
        if (isBeingUsed)
        {
            // Luồng A-4: Chỉ cập nhật trạng thái Ngừng kinh doanh
            if (command.Mode == "DISCONTINUE_ONLY")
            {
                package.Discontinue();
                packageRepository.Update(package);
                await unitOfWork.SaveChangesAsync(cancellationToken);

                return Result.Success(await MapDetailAsync(package, cancellationToken));
            }

            // Luồng A-3: Ngừng kinh doanh gói cũ + Tạo gói mới có liên kết phả hệ PreviousPackageId
            if (command.Mode == "DISCONTINUE_AND_CREATE")
            {
                package.Discontinue();
                packageRepository.Update(package);

                var newPackage = MembershipPackage.Create(
                    command.Name,
                    command.Price,
                    command.DurationMonths,
                    command.EffectiveStartDate,
                    command.EffectiveEndDate,
                    command.MaxCheckins,
                    command.Description,
                    previousPackageId: package.Id); // Lưu vết phả hệ gói tiền nhiệm

                await packageRepository.AddAsync(newPackage, cancellationToken);

                var saveSucceeded = false;
                for (var attempt = 1; attempt <= MaxSaveRetries; attempt++)
                {
                    try
                    {
                        await unitOfWork.SaveChangesAsync(cancellationToken);
                        saveSucceeded = true;
                        break;
                    }
                    catch (DbUpdateException ex) when (IsUniqueConstraintViolation(ex) && attempt < MaxSaveRetries)
                    {
                        newPackage.RegenerateCode(MembershipPackage.GenerateRandomCode());
                    }
                }

                if (!saveSucceeded)
                {
                    return Result.Failure<PackageDetailDto>(
                        Error.Conflict("Package.CodeCollision", "Không thể sinh mã cho gói tập mới. Vui lòng thử lại."));
                }

                return Result.Success(await MapDetailAsync(newPackage, cancellationToken, previousCode: package.PackageCode));
            }

            // Nếu chưa chọn Mode: Trả về cảnh báo 409 Conflict để frontend kích hoạt Dialog A-3 / A-4
            return Result.Failure<PackageDetailDto>(
                Error.Conflict("Package.InUse", "Gói tập này đang có người sử dụng. Bạn có muốn cập nhật trạng thái 'Ngừng kinh doanh' và tạo 1 gói tập mới với các trường thông tin đã sửa không?"));
        }

        // Trường hợp KHÔNG có người dùng: Cập nhật trực tiếp
        package.UpdateInfo(
            command.Name,
            command.Price,
            command.DurationMonths,
            command.EffectiveStartDate,
            command.EffectiveEndDate,
            command.Status,
            command.MaxCheckins,
            command.Description);

        packageRepository.Update(package);
        await unitOfWork.SaveChangesAsync(cancellationToken);

        return Result.Success(await MapDetailAsync(package, cancellationToken));
    }

    private static bool IsUniqueConstraintViolation(DbUpdateException ex) =>
        ex.InnerException is PostgresException pgEx && pgEx.SqlState == PostgresErrorCodes.UniqueViolation;

    private async Task<PackageDetailDto> MapDetailAsync(MembershipPackage pkg, CancellationToken ct, string? previousCode = null)
    {
        var activeCount = await packageRepository.CountActiveMembersAsync(pkg.Id, ct);
        return new PackageDetailDto
        {
            Id = pkg.Id,
            PackageCode = pkg.PackageCode,
            Name = pkg.Name,
            Price = pkg.Price,
            DurationMonths = pkg.DurationMonths,
            DurationDays = pkg.DurationDays,
            MaxCheckins = pkg.MaxCheckins,
            Description = pkg.Description,
            Status = pkg.Status,
            EffectiveStartDate = pkg.EffectiveStartDate,
            EffectiveEndDate = pkg.EffectiveEndDate,
            PreviousPackageId = pkg.PreviousPackageId,
            PreviousPackageCode = previousCode,
            ActiveMembersCount = activeCount,
            CreatedDate = pkg.CreatedDate,
            LastModifiedDate = pkg.LastModifiedDate
        };
    }
}
```

```csharp
        if (result.IsSuccess)
        {
            return Ok(new { success = true, data = result.Value });

        return result.Error.Code switch

        if (result.IsSuccess)
        {
            return Ok(new { success = true, data = result.Value, message = "Cập nhật gói tập thành công." });

        return result.Error.Code switch
        {
            "Package.NotFound" => NotFound(new { error = result.Error }),
            "Package.InUse" => Conflict(new { error = result.Error, canDiscontinueAndCreate = true }),
            _ => BadRequest(new { error = result.Error })

            "Package.CannotDeleteInUse" => Conflict(new { error = result.Error, canDiscontinue = true }),
```
